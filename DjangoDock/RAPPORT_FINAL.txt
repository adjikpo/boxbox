🔧 COMMANDES MAKEFILE (INCLUSES)
================================================================================

INITIALISATION:
  make init           # Build + migrate + createsuperuser

DOCKER:
  make build          # Builder les images
  make up             # Démarrer (background)
  make down           # Arrêter
  make restart        # Redémarrer
  make ps             # État
  make dev            # Mode dev (logs en direct)

DJANGO:
  make migrate        # Appliquer migrations
  make migrations     # Créer migrations
  make createsuperuser # Admin Django
  make collectstatic  # Fichiers statiques
  make shell          # Django shell
  make test           # Exécuter tests

ACCÈS:
  make bash-web       # Shell du container
  make bash-db        # PostgreSQL psql

LOGS:
  make logs           # Tous les services
  make logs-web       # Django seulement
  make logs-db        # PostgreSQL seulement

NETTOYAGE:
  make clean          # Arrêter + supprimer volumes
  make help           # Afficher l'aide
